  <script>
    lucide.createIcons();

    async function sendToAI() {
      const transcript = document.getElementById('userTranscript').value.trim();
      const question = document.getElementById('questionText').innerText;
      const btn = document.getElementById('analyzeBtn');
      const resultPanel = document.getElementById('aiResult');

      if (!transcript) {
        alert("Vui lòng nhập câu trả lời của bạn!");
        return;
      }

      // Đổi trạng thái nút
      btn.disabled = true;
      btn.innerHTML = `<i data-lucide="loader-2" class="w-4 h-4 animate-spin"></i> <span>Hệ thống đang xử lý...</span>`;
      lucide.createIcons();
      resultPanel.classList.add('hidden');

      try {
        const response = await fetch('/api/analyze', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            userTranscript: transcript,
            targetBand: "7.0",
            question: question
          })
        });

        const data = await response.json();

        if (data.error) {
          alert("Lỗi máy chủ: " + data.error);
        } else {
          document.getElementById('bandBadge').innerText = "Band " + data.estimatedBand;
          document.getElementById('aiStrengths').innerText = data.strengths;
          document.getElementById('aiImprovements').innerText = data.improvements;
          document.getElementById('aiBetterPhrasing').innerText = `"${data.betterPhrasing}"`;
          resultPanel.classList.remove('hidden');
        }
      } catch (err) {
        alert("Không thể kết nối với API. Vui lòng kiểm tra lại cấu hình trên Vercel.");
        console.error(err);
      } finally {
        btn.disabled = false;
        btn.innerHTML = `<i data-lucide="sparkles" class="w-4 h-4"></i> <span>Phân Tích Cốt Lõi Bằng AI</span>`;
        lucide.createIcons();
      }
    }
  </script>
