async function sendToAI() { const userText = document.getElementById('userTranscript').value; const analyzeBtn = document.getElementById('analyzeBtn'); if (!userText.trim()) { alert("Vui lòng nhập nội dung trả lời trước!"); return; } analyzeBtn.innerText = "Hệ thống đang xử lý..."; analyzeBtn.disabled = true; try { const response = await fetch('/api/analyze', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: userText }) }); const result = await response.json(); if (!response.ok) { // Đọc trực tiếp chuỗi lỗi đã bóc tách từ Server alert("Lỗi máy chủ: " + (result.error || "Không xác định")); return; } // Xử lý hiển thị kết quả từ Gemini const aiText = result.candidates[0].content.parts[0].text; console.log("AI Phân tích:", aiText); alert("Phân tích thành công! Xem kết quả bên dưới."); } catch (err) { alert("Lỗi kết nối: " + err.message); } finally { analyzeBtn.innerText = "Phân Tích Cốt Lõi Bằng AI"; analyzeBtn.disabled = false; } }
